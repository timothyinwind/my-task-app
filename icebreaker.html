<script>

  // 預設卡片庫

  const defaultCards = [

    { cat: "輕鬆", q: "如果你可以瞬間學會一項技能，你希望是什麼？" },

    { cat: "趣味", q: "你吃過最奇怪但意想不到好吃的食物是什麼？" },

    { cat: "深度", q: "當你覺得焦慮或壓力大時，最能讓你安心的放鬆方式是什麼？" },

    { cat: "職場", q: "你最喜歡的工作環境或氛圍具備什麼要素？" },

    { cat: "生活", q: "如果明天突然多出一整天的假，且不能用手機，你會想做什麼？" },

    { cat: "音樂", q: "有沒有哪一首歌，是你心情不好時聽一定能被療癒的？" },

    { cat: "秩序", q: "你生活中有沒有什麼不可妥協的「小堅持」或「儀式感」？" }

  ];

  function getCards() {

    const data = localStorage.getItem('icebreaker_cards');

    return data ? JSON.parse(data) : defaultCards;

  }

  function saveCards(cards) {

    localStorage.setItem('icebreaker_cards', JSON.stringify(cards));

    updateCount();

  }

  function updateCount() {

    const cards = getCards();

    document.getElementById('countText').innerText = `共 ${cards.length} 張卡片`;

  }

  function drawCard() {

    const cards = getCards();

    if (cards.length === 0) {

      document.getElementById('tagText').innerText = "提示";

      document.getElementById('qText').innerText = "目前沒有任何卡片，請先點擊「新增」！";

      return;

    }

    const randomIndex = Math.floor(Math.random() * cards.length);

    const card = cards[randomIndex];

    document.getElementById('tagText').innerText = card.cat || "破冰問題";

    document.getElementById('qText').innerText = card.q;

  }

  function openAddModal() { document.getElementById('addModal').style.display = 'flex'; }

  function openManageModal() { renderManageList(); document.getElementById('manageModal').style.display = 'flex'; }

  function closeModal(id) { document.getElementById(id).style.display = 'none'; }

  function saveNewCard() {

    const cat = document.getElementById('newCategory').value.trim() || "常規";

    const q = document.getElementById('newQuestion').value.trim();

    if (!q) { alert("請輸入問題內容！"); return; }

   

    const cards = getCards();

    cards.push({ cat, q });

    saveCards(cards);

    document.getElementById('newCategory').value = '';

    document.getElementById('newQuestion').value = '';

    closeModal('addModal');

    drawCard();

  }

  function renderManageList() {

    const cards = getCards();

    const list = document.getElementById('qList');

    list.innerHTML = cards.map((c, i) => `

      <div class="q-item">

        <span><b>[${c.cat}]</b> ${c.q}</span>

        <button style="background:none; border:none; color:#e57373; cursor:pointer;" onclick="deleteCard(${i})">刪除</button>

      </div>

    `).join('');

  }

  function deleteCard(index) {

    const cards = getCards();

    cards.splice(index, 1);

    saveCards(cards);

    renderManageList();

  }

  function resetDefault() {

    if (confirm("確定要重置為預設的卡片庫嗎？")) {

      saveCards(defaultCards);

      renderManageList();

      drawCard();

    }

  }

  function exportData() {

    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(getCards(), null, 2));

    const dlAnchorElem = document.createElement('a');

    dlAnchorElem.setAttribute("href", dataStr);

    dlAnchorElem.setAttribute("download", "icebreaker_cards.json");

    dlAnchorElem.click();

  }

  function importData() {

    const jsonStr = prompt("請貼入破冰問題的 JSON 字串：");

    if (!jsonStr) return;

    try {

      const cards = JSON.parse(jsonStr);

      if (Array.isArray(cards)) {

        saveCards(cards);

        renderManageList();

        drawCard();

        alert("匯入成功！");

      }

    } catch(e) {

      alert("JSON 格式錯誤！");

    }

  }

  // 初始化

  updateCount();

</script>
