    function updateCard(type, index, key, value) {

      appData[type][index][key] = value;

      saveData();

    }

    function addNewCard(type) {

      const isTask = type === 'tasks';

      const newId = (isTask ? 'T' : 'R') + String(appData[type].length + 1).padStart(2, '0');

      appData[type].push(isTask

        ? { id: newId, category: '自訂', title: '新任務', desc: '輸入內容...', hint: '啟動提示...' }

        : { id: newId, category: '自訂', title: '新獎勵', desc: '輸入內容...', focus: '體驗重點...' }

      );

      saveData();

    }

    function deleteCard(type, index) {

      if (confirm('確定刪除這張卡片？')) {

        appData[type].splice(index, 1);

        saveData();

      }

    }

    function exportJSON() {

      navigator.clipboard.writeText(JSON.stringify(appData, null, 2));

      alert('已複製 JSON 至剪貼簿！');

    }

    function exportMarkdown() {

      let md = `## 任務卡庫\n| 編號 | 類別 | 任務名稱 | 10-15分鐘執行內容 | 最小啟動提示 |\n| :--- | :--- | :--- | :--- | :--- |\n`;

      appData.tasks.forEach(t => { md += `| **${t.id}** | ${t.category} | **${t.title}** | ${t.desc} | ${t.hint || ''} |\n`; });

      md += `\n## 獎勵卡庫\n| 編號 | 類別 | 獎勵名稱 | 即時獎勵內容 | 體驗重點 |\n| :--- | :--- | :--- | :--- | :--- |\n`;

      appData.rewards.forEach(r => { md += `| **${r.id}** | ${r.category} | **${r.title}** | ${r.desc} | ${r.focus || ''} |\n`; });

      navigator.clipboard.writeText(md);

      alert('已複製 Markdown 表格至剪貼簿！');

    }

    function importJSON() {

      const input = document.getElementById('import-textarea').value.trim();

      if (!input) return alert('請先貼入資料！');

      try {

        const parsed = JSON.parse(input);

        if (parsed.tasks && parsed.rewards) {

          appData = parsed;

          saveData();

          alert('匯入成功！');

          document.getElementById('import-textarea').value = '';

          switchTab('draw');

          drawRandomCards();

        }

      } catch (e) {

        alert('格式不正確！');

      }

    }
