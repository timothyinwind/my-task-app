<script>

  function renderLinks() {

    const links = JSON.parse(localStorage.getItem('portal_custom_links') || '[]');

    const list = document.getElementById('linkList');

    if (links.length === 0) {

      list.innerHTML = '<div style="color:#95a5a6; font-size:0.85rem; text-align:center; padding:10px;">尚未新增自訂連結</div>';

      return;

    }

    list.innerHTML = links.map((item, index) => `

      <div class="custom-item">

        <a href="${item.url}" target="_blank">${item.title}</a>

        <button class="del-btn" onclick="deleteLink(${index})">刪除</button>

      </div>

    `).join('');

  }

  function addCustomLink() {

    const title = prompt('請輸入連結名稱：');

    if (!title) return;

    let url = prompt('請輸入網址 (如 https://google.com)：');

    if (!url) return;

    if (!url.startsWith('http://') && !url.startsWith('https://')) {

      url = 'https://' + url;

    }

    const links = JSON.parse(localStorage.getItem('portal_custom_links') || '[]');

    links.push({ title, url });

    localStorage.setItem('portal_custom_links', JSON.stringify(links));

    renderLinks();

  }

  function deleteLink(index) {

    const links = JSON.parse(localStorage.getItem('portal_custom_links') || '[]');

    links.splice(index, 1);

    localStorage.setItem('portal_custom_links', JSON.stringify(links));

    renderLinks();

  }

  renderLinks();

</script>
